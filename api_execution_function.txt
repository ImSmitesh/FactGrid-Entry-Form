document.getElementById("monasteriesForm").onsubmit = async e => {
    e.preventDefault();

    const fd = new FormData(e.target);
    const fields = ["P131", "P2", "P746", "P662", "P471", "P76"]; /*"P208"*/
    
    let qid = "";

    try {
        // const res = await fetch("/create-empty-item", {
        //     method: "POST",
        //     headers: { "Content-Type": "application/json" }
        // });
        
        // if (!res.ok) throw new Error("Failed to reach server to create QID.");
        // const result = await res.json();
        
        // if (!result.id) throw new Error("API failed to return a valid QID.");
        // qid = result.id;
        qid = 1

        /* ===============================
            2. Build Monastery V1 Command
        =============================== */
        const data = {};
        fields.forEach(field => {
            let rawValue = fd.get(field);
            if (!rawValue) return;

            // Use the formatter to handle quotes for P662, P471, P76, etc.
            let formattedMain = formatValueByPid(field, rawValue);
            
            const qualifiers = buildQualifierParts(field);
            data[field] = qualifiers ? `${formattedMain} ${qualifiers}` : formattedMain;
        });

        ["L", "D", "A"].forEach(prefix => {
            const langs = fd.getAll(`${prefix}_lang[]`);
            const vals = fd.getAll(`${prefix}_value[]`);
            langs.forEach((lang, i) => {
                if (!lang || !vals[i]) return;
                const key = `${prefix}${lang.trim()}`;
                if (!data[key]) data[key] = [];
                data[key].push(`"${vals[i]}"`);
            });
        });

        let command = "";
        Object.entries(data).forEach(([k, v]) => {
            if (Array.isArray(v)) {
                v.forEach(x => { command += `${qid}\t${k}\t${x}\n`; });
            } else {
                command += `${qid}\t${k}\t${v}\n`;
            }
        });

        if (!command) throw new Error("No data provided.");

        /* ===============================
            3. PASS TO COMPLEX FORM
        =============================== */
        // Store the V1 command and the new QID for the next form
        localStorage.setItem("pending_monastery_v1", command);
        localStorage.setItem("last_monastery_qid", qid);

        console.log("Monastery V1 Command Prepared for QID:", qid);
        console.log("Command Content:\n", command);
        
        showStatus(`Success! New QID ${qid} created. Please click 'Complexes Form' to proceed.`);

    } catch (err) {
        console.error("Critical Failure:", err.message);
        showStatus(`Error: ${err.message}`, true);
    }
};
